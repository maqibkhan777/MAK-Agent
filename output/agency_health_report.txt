import os
import json

def merge_json_files(directory):
    """
    Merge all .json files in the specified directory into a single dictionary.
    
    Args:
        directory (str): The directory to scan for .json files.
    
    Returns:
        dict: A single dictionary containing the data from all .json files.
    """
    
    merged_dict = {}
    for filename in os.listdir(directory):
        if filename.endswith(".json"):
            file_path = os.path.join(directory, filename)
            with open(file_path, 'r') as file:
                merged_dict.update(json.load(file))
    
    return merged_dict

# Example usage:
directory = '/path/to/json/files'
merged_dict = merge_json_files(directory)
print(merged_dict)
